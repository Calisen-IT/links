# Calisen - Autopilot device registration
# Run from OOBE (Shift+F10):
#   powershell -ep bypass -c "irm https://links.calisen.com/ap.txt | iex"

[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
Set-ExecutionPolicy -Scope Process -ExecutionPolicy Bypass -Force
Install-PackageProvider -Name NuGet -MinimumVersion 2.8.5.201 -Force | Out-Null
Set-PSRepository -Name PSGallery -InstallationPolicy Trusted

# Install all modules up front so nothing tries to update a module that's already loaded
Write-Host "Installing modules (this can take a minute)..." -ForegroundColor Cyan
Install-Module -Name Microsoft.Graph.Authentication, Microsoft.Graph.Groups, Microsoft.Graph.Identity.DirectoryManagement, WindowsAutopilotIntune -Force

Install-Script -Name Get-WindowsAutopilotInfo -Force
$env:Path += ";C:\Program Files\WindowsPowerShell\Scripts"

try {
    Get-WindowsAutopilotInfo -Online -AddToGroup "Auto Pilot Enrollment Devices" -Assign
}
finally {
    Disconnect-MgGraph -ErrorAction SilentlyContinue
}

Write-Host "Done. Restarting in 10 seconds..." -ForegroundColor Green
Start-Sleep 10
Restart-Computer -Force
